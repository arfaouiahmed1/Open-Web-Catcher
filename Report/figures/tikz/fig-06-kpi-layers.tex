\begin{tikzpicture}[
  line cap=round,
  line join=round,
  node distance=\kpiGapY and \kpiGapX,
  panel/.style={draw=gray!32, fill=gray!2, rounded corners=10pt,
    line width=0.55pt, inner sep=\kpiPanelPad},
  card/.style={draw=accent!58, fill=white, rounded corners=7pt,
    line width=0.62pt, align=left, font=\scriptsize,
    text width=\kpiCardW, minimum height=\kpiCardH, inner sep=\kpiPad},
  outcome/.style={card, draw=accent!68, fill=blue!5},

  title/.style={font=\scriptsize\bfseries, fill=white, inner xsep=4pt,

  formula/.style={font=\tiny\ttfamily, text=black!64},
  note/.style={font=\tiny, text=black!62},
  arr/.style={-{Latex[length=2.0mm,width=1.45mm]}, accent!72,

\node[outcome] (strict)
  {\textbf{Strict success}\\[-1pt]\kpiFormula{success / terminal}\\[-1pt]\kpiNote{headline dashboard rate}};
\node[outcome, right=of strict] (productive)
  {\textbf{Productive evidence}\\[-1pt]\kpiFormula{(success + partial) / terminal}\\[-1pt]\kpiNote{reviewable full or partial output}};
\node[blocker, right=of productive] (blocker)
  {\textbf{External blocker}\\[-1pt]\kpiFormula{blocked / terminal}\\[-1pt]\kpiNote{site, hosting, player, or stream unavailable}};

\node[diagnostic, below=of strict] (agent)
  {\textbf{Agent-adjusted diagnostic}\\[-1pt]\kpiFormula{(productive + blocker) / non-cancelled}\\[-1pt]\kpiNote{triage signal, not a success rate}};
\node[diagnostic, right=of agent] (tool)
  {\textbf{Tool reliability}\\[-1pt]\kpiFormula{successful tools / observed tools}\\[-1pt]\kpiNote{separates MCP/browser stability from page outcome}};
\node[diagnostic, right=of tool] (cost)
  {\textbf{Runtime cost}\\[-1pt]\kpiFormula{tokens + tools + latency + spend}\\[-1pt]\kpiNote{checks whether batch evaluation can scale}};
\node[diagnostic, below=of tool] (repeat)
  {\textbf{Regression retention}\\[-1pt]\kpiFormula{accepted again / previously accepted}\\[-1pt]\kpiNote{guards against prompt or browser regressions}};

\begin{scope}[on background layer]
  \node[panel, fit=(strict) (productive) (blocker)] (outcomePanel) {};
  \node[panel, fit=(agent) (tool) (cost) (repeat)] (diagnosticPanel) {};
